user and the experts: each delegate the model emits is parsed, validated and answered with a
mocked expert result authored here (album lists, search-picture items with paths and times,

  python -I simulate.py [--runs 2] [--start-run 1] [--only S1,S4] [--budget 120]
  python -I simulate.py --replay results/sim/X.json --upto N --as-run R --budget B
      feeds the first N recorded assistant turns back (no API calls), re-runs the mocks and the

  python -I simulate.py --reeval     recompute all checks on saved runs, no API calls
Writes results/sim/<scenario>_r<n>.json and results/sim/summary.json.
The API key is read at call time by common.chat and is never printed or stored.

def build_system_prompt():
    skill = (SKILL_DIR / "SKILL.md").read_text(encoding="utf-8")
    links = re.findall(r"\]\((references/[^)]+\.md)\)", skill)
    seen = []
    for link in links:
        if link not in seen:
            seen.append(link)
    parts = [PREAMBLE, "===== SKILL.md =====", skill]
    for link in seen:
        parts += [f"===== {link} =====", (SKILL_DIR / link).read_text(encoding="utf-8")]
    return "\n\n".join(parts), seen


def load_vision_prompts():
    text = (SKILL_DIR / "references" / "vision-prompts.md").read_text(encoding="utf-8")
    out = {}
    for key, head in (("brief", "## Brief prompt"), ("rich", "## Rich prompt"),
                      ("question", "## Question prompt")):
        block = text.split(head, 1)[1].split("\n## ", 1)[0]
        paras = [p.strip() for p in block.split("\n\n") if p.strip()]
        out[key] = paras[1]  # paras[0] is the rest of the heading line
    return out

# --------------------------------------------------------------------------- parsing


def norm(s):
    return (s or "").replace(APO, "'")


def strip_think(t):
    return re.sub(r"<think>.*?</think>", "", t or "", flags=re.S).strip()


def parse_delegates(text):
    """Return (list of delegate dicts, protocol notes)."""
    notes = []
    starts = [m.start() for m in re.finditer(r"delegate\s+--target", text)]
    if not starts:
        return [], notes
    if len(starts) > 1:
        notes.append(f"{len(starts)} delegate commands in one message")
    out = []
    for i, s in enumerate(starts):
        seg = text[s: starts[i + 1] if i + 1 < len(starts) else len(text)]
        pre = text[max(0, s - 12):s]
        if "```" in pre or "`" in pre[-2:]:
            notes.append("delegate inside a code block")
        tail = re.sub(r"\s*```\s*$", "", seg.rstrip())
        m = re.match(r"delegate\s+--target\s+(\S+)\s+--instruction\s+'(.*)'", tail, re.S)
        d = {"raw": seg, "target": None, "instruction": None, "issues": [], "after": ""}
        if not m:
            m2 = re.match(r"delegate\s+--target\s+(\S+)", tail)
            d["target"] = m2.group(1) if m2 else None
            d["issues"].append("format: no single-quoted --instruction found")
            out.append(d)
            continue
        d["target"], d["instruction"] = m.group(1), m.group(2)
        after = tail[m.end():].strip()
        if after:
            d["after"] = after[:200]
            notes.append("text after the delegate command")
        out.append(d)
    return out, notes


def hygiene(target, ins, step):
    issues = []
    if target not in AGENTS:
        issues.append(f"target {target} is not a registered agent")
    if ins is None:
        return issues
    if "'" in ins:
        n = ins.count("'")
        issues.append(f"{n} straight apostrophe(s) inside the instruction")
    if '"' in ins:
        issues.append("double quote inside the instruction")
    for sign, name in (("`", "backtick"), ("--", "two hyphens"), (">", "greater-than"),
                       ("<", "less-than"), ("|", "vertical bar"), ("$(", "dollar-parenthesis"),
                       ("&&", "two ampersands")):
        if sign in ins:
            issues.append(f"{name} inside the instruction")
    slots = re.findall(r"\{[^{}\n]{1,80}\}", ins)
    if slots:
        issues.append("unresolved slot(s): " + ", ".join(sorted(set(slots))[:5]))
    if "\n" in ins and step not in ("7", "Q3"):
        issues.append("line break in a one-line instruction")
    for leak in ("Garden party", "7f3c2a10", "IMG_2041", "IMG_2047", "IMG_2101", "Robin",
                 "2026-06-14"):
        if leak in ins:
            issues.append(f"example value copied: {leak}")
    return issues


def trace_values(ins, corpus):
    """Paths, UUIDs and times in the instruction must appear in earlier mocked results or user turns."""
    n_ins, n_cor = norm(ins), norm(corpus)
    bad = []
    for p in re.findall(r"/volume\d+/[^;\n]*?\.(?:jpe?g|png|heic|txt)", n_ins):
        if p not in n_cor:
            bad.append(p)
    for u in re.findall(r"\b[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}\b", n_ins):
        if u not in n_cor:
            bad.append(u)
    for t in re.findall(r"\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}", n_ins):
        if t not in n_cor:
            bad.append(t)
    m = re.search(r"folder = (/volume[^;]*?)(?:;|\.\s|$)", n_ins)
    if m and m.group(1).rstrip(". ") not in n_cor:
        bad.append(m.group(1))
    return bad


def classify(target, ins):
    l = (ins or "").lower()
    if target == "photo-agent":
        if "vision-vlm" in l:
            return "Q2" if ("asks a question about one family photo" in l or "max tokens 80" in l) else "4"
        if "still photos" in l or "selection rule" in l:
            return "3"
        if "album" in l:
            return "1"
    if target == "file-agent":
        if "first 300" in l:
            return "8"
        if "start with the name value" in l or "names of its files" in l:
            return "6"
        if "photo number" in l:
            return "Q1"
        if "listening letter" in l:
            return "2"
    if target == "knowledge-agent":
        return "7" if "letter text" in l else "Q3"
    return "?"


def values_part(ins):
    m = re.search(r"Values:(.*)$", ins or "", re.S)
    return m.group(1).strip() if m else ""


def get_value(ins, key):
    m = re.search(re.escape(key) + r"\s*=\s*(.*?)(?:;\s|\.$|$)", values_part(ins), re.S)
    return m.group(1).strip().rstrip(".") if m else None

def ok(data, actions):
    return {"status": "completed", "actionsRun": actions, "data": data}


def find_photos(ins, pool):
    """Match P numbers and paths in a vision instruction against the photos the expert returned."""
    n = norm(values_part(ins))
    found, unknown = [], []
    for m in re.finditer(r"P(\d+)\D{0,6}?(/volume\d+/[^;\n]*?\.jpe?g)", n, re.I):
        p, path = int(m.group(1)), m.group(2)
        hit = next((ph for ph in pool if ph["path"] == path), None)
        (found if hit else unknown).append((p, path, hit))
    return found, unknown

# --------------------------------------------------------------------------- one run

LOCK = threading.Lock()
BUDGET = {"left": 120, "used": 0}


def call_model(messages):
    with LOCK:
        if BUDGET["left"] <= 0:
            return None, "budget exhausted"
        BUDGET["left"] -= 1
    r = chat_retry(MODEL, messages, TEMPERATURE, MAX_TOKENS, extra=THINKING_OFF)
    with LOCK:
        BUDGET["used"] += r["attempts"]
        if r["attempts"] > 1:
            BUDGET["left"] -= r["attempts"] - 1
    if not r["ok"]:
        return None, "; ".join(r["errors"])[:300]
    e = extract(r["response"])
    return {"content": strip_think(e["content"]), "finish": e["finish_reason"], "usage": e["usage"],
            "seconds": r["seconds"]}, None


def run_scenario(sid, run_no, system_prompt, replay=None):
    """replay: list of recorded assistant texts that are fed back instead of calling the model.
    Mocks and the simulated user are run again on them, so a harness fix does not cost API calls."""
    sc = SCENARIOS[sid]
    ctx = {}
    messages = [{"role": "system", "content": system_prompt}, {"role": "user", "content": sc["first"]}]
    log = {"scenario": sid, "run": run_no, "title": sc["title"], "turns": [{"role": "user", "content": sc["first"]}],
           "delegates": [], "protocol": [], "error": None, "calls": 0, "replayed_turns": len(replay or [])}
    corpus = sc["first"]
    replay = list(replay or [])
    for _ in range(sc["cap"] + len(replay)):
        if replay:
            res, err = {"content": replay.pop(0), "finish": "replayed"}, None
        else:
            res, err = call_model(messages)
            log["calls"] += 1
        if err:
            log["error"] = err
            break
        text = res["content"] or ""
        log["turns"].append({"role": "assistant", "content": text, "finish": res["finish"]})
        messages.append({"role": "assistant", "content": text})
        dels, notes = parse_delegates(text)
        log["protocol"] += notes
        if dels:
            d = dels[0]
            step = classify(d["target"], d["instruction"])
            issues = d["issues"] + hygiene(d["target"], d["instruction"], step)
            untraced = trace_values(d["instruction"] or "", corpus)
            if untraced:
                issues.append("values not found in earlier results: " + "; ".join(untraced[:4]))
            try:
                result = mock(sid, ctx, step, d["target"], d["instruction"] or "")
            except Exception as ex:  # a harness bug must not end the whole batch
                result = {"status": "failed", "reason": "expert error"}
                log["protocol"].append(f"harness mock error: {type(ex).__name__}: {ex}")
            if step == "8" and result.get("status") == "completed":
                ctx["step8_done"] = True
            if step == "Q3" and result.get("status") == "completed":
                ctx["q3_done"] = True
            log["delegates"].append({"turn": len(log["turns"]) - 1, "step": step, "target": d["target"],
                                     "instruction": d["instruction"], "issues": issues,
                                     "result": result, "chat_before": text[: text.find("delegate")].strip()[:400]})
            msg = f"EXPERT RESULT from {d['target']}:\n" + json.dumps(result, ensure_ascii=False, indent=1)
            corpus += "\n" + msg
            log["turns"].append({"role": "user", "content": msg})
            messages.append({"role": "user", "content": msg})
            if sid == "S3" and step == "3":
                break
            continue
        nxt = user_reply(sid, ctx, text)
        if nxt is None:
            break
        corpus += "\n" + nxt
        log["turns"].append({"role": "user", "content": nxt})
        messages.append({"role": "user", "content": nxt})
    log["ctx"] = {k: v for k, v in ctx.items() if k != "saved_text"}
    log["saved_text"] = ctx.get("saved_text")
    log["checks"] = evaluate(sid, log)
    log["pass"] = all(c["ok"] for c in log["checks"] if c["critical"])
    OUT.mkdir(parents=True, exist_ok=True)
    (OUT / f"{sid}_r{run_no}.json").write_text(json.dumps(log, ensure_ascii=False, indent=1), encoding="utf-8")
    return log

GENDER = re.compile(r"\b(man|woman|men|women|boy|girl|he|she|his|her|hombre|mujer|niña|chico|chica)\b", re.I)
NAMES = re.compile(r"\b(Ana|Grandpa|Rosa|Toby|Tomás|Carmen|abuela)\b")
WEEKDAY = re.compile(r"\b(Monday|Tuesday|Wednesday|Thursday|Friday|Saturday|Sunday|lunes|martes|miércoles|jueves|viernes|sábado|domingo)\b", re.I)


def chk(name, okv, detail="", critical=True):
    return {"name": name, "ok": bool(okv), "detail": detail, "critical": critical}


def assistant_texts(log):
    return [t["content"] for t in log["turns"] if t["role"] == "assistant"]


def previews(log):
    return [t for t in assistant_texts(log) if "preview:" in t.lower() or "nothing is saved yet" in t.lower()
            or "vista previa" in t.lower()]


def entries(letter):
    """Split a letter into {N: [lines]} by the Photo N of M. / Foto N de M. lines."""
    out, cur = {}, None
    for line in letter.splitlines():
        m = re.match(r"\s*(?:Photo|Foto)\s+(\d+)\s+(?:of|de)\s+(\d+)\.?\s*$", line)
        if m:
            cur = int(m.group(1))
            out[cur] = []
            continue
        if cur is not None:
            if not line.strip():
                cur = None
            else:
                out[cur].append(line.strip())
    return out


def letter_checks(letter, exp, prefix):
    c = []
    L = norm(letter)
    e = entries(letter)
    c.append(chk(f"{prefix}: title line Photos from album", re.search(r"^(Photos from|Fotos de|Fotos del álbum) " + re.escape(norm(exp["album"])), L, re.M), L.splitlines()[0][:80] if L else "empty"))
    c.append(chk(f"{prefix}: {exp['M']} numbered entries", sorted(e) == list(range(1, exp["M"] + 1)), f"found {sorted(e)}"))
    ai = [ln for lines in e.values() for ln in lines if re.match(r"(AI guess|Suposición|Conjetura|IA)", ln)]
    bad = [ln for ln in ai if NAMES.search(ln) or GENDER.search(ln)]
    c.append(chk(f"{prefix}: AI guess lines hold no name or gender word", not bad, "; ".join(bad)[:200]))
    empty = [ln for lines in e.values() for ln in lines if re.search(r":\s*(none|ninguna?|nada)\.?$", ln, re.I)]
    c.append(chk(f"{prefix}: lines with no content left out (no ': none')", not empty, "; ".join(sorted(set(empty)))[:160], critical=False))
    c.append(chk(f"{prefix}: no weekday", not WEEKDAY.search(L), (WEEKDAY.search(L) or [""])[0]))
    c.append(chk(f"{prefix}: plain text, no markup", not re.search(r"^#|\*\*|^\s*[-*] ", L, re.M), "", critical=False))
    for anchor in ("For the helper: photo files", "Selection rule: 1", f"Newest photo checked: {exp['newest']}"):
        c.append(chk(f"{prefix}: anchor {anchor}", anchor in L))
    for i, ph in enumerate(exp["photos"], 1):
        if f"Photo {i}: {norm(ph['path'])}" not in L:
            c.append(chk(f"{prefix}: Photo {i}: real path line", False, ph["path"]))
            break
    else:
        c.append(chk(f"{prefix}: Photo N: real path lines", True))
    tail = L.rstrip().splitlines()[-1] if L.strip() else ""
    c.append(chk(f"{prefix}: photo files section is last", tail.startswith(f"Photo {exp['M']}:"), tail[:80]))
    for n, reason in exp.get("not_described", {}).items():
        lines = " ".join(e.get(n, []))
        c.append(chk(f"{prefix}: photo {n} Not described with reason", ("not described" in lines.lower() or "no se describ" in lines.lower()) and reason.lower() in lines.lower(), lines[:150]))
        c.append(chk(f"{prefix}: photo {n} has no AI guess", "AI guess" not in lines, lines[:150]))
    for n, words in exp.get("words", {}).items():
        lines = " ".join(e.get(n, []))
        c.append(chk(f"{prefix}: photo {n} Words in the photo from OCR", words in lines, lines[-160:]))
        c.append(chk(f"{prefix}: photo {n} OCR words not marked (AI reading)", "(AI reading)" not in lines, lines[-160:], critical=False))
    for n in exp.get("no_long_number", []):
        lines = " ".join(e.get(n, []))
        c.append(chk(f"{prefix}: photo {n} drops numbers over 4 digits", not re.search(r"\d{5,}", lines), lines[-120:]))
    for n in exp.get("file_date", []):
        lines = " ".join(e.get(n, []))
        c.append(chk(f"{prefix}: photo {n} When says from the file date", "file date" in lines or "fecha del archivo" in lines, lines[:120]))
    for n, place in exp.get("where", {}).items():
        lines = " ".join(e.get(n, []))
        c.append(chk(f"{prefix}: photo {n} Where {place}", place in lines, lines[:120], critical=False))
    for n, frag in exp.get("family", {}).items():
        lines = e.get(n, [])
        fam = [i for i, ln in enumerate(lines) if ln.startswith("Family says")]
        aig = [i for i, ln in enumerate(lines) if ln.startswith("AI guess")]
        c.append(chk(f"{prefix}: photo {n} Family says line", fam and frag in norm(lines[fam[0]]), " | ".join(lines)[:200]))
        if fam and aig:
            c.append(chk(f"{prefix}: photo {n} Family says before AI guess", fam[0] < aig[0]))
    if exp.get("no_family"):
        fam_lines = [ln for lines in e.values() for ln in lines if ln.startswith("Family says")]
        c.append(chk(f"{prefix}: no Family says line before any note", not fam_lines, "; ".join(fam_lines)[:150]))
    return c

        c.append(chk("no vision call for an identity question", "Q2" not in steps and "4" not in steps, str(steps)))

def vision_prompt_checks(groups, mode, language):
    c = []
    if not groups:
        return [chk(f"{mode} vision call made", False)]
    exp = VISION_PROMPTS[mode].replace("{language}", language)
    tokens = {"brief": 90, "rich": 170, "question": 80}[mode]
    for k, d in enumerate(groups):
        ins = d["instruction"] or ""
        m = re.search(r"Prompt begins:\s*(.*?)\s*Prompt ends\.", ins, re.S)
        sent = " ".join((m.group(1) if m else "").split())
        if mode == "question":
            pat = re.escape(" ".join(exp.split())).replace(re.escape("{question}"), "(.+?)")
            same = bool(re.fullmatch(pat, sent))
            ratio = difflib.SequenceMatcher(None, re.sub(r"The question is: .*?\. Step 1", "", sent), re.sub(r"The question is: .*?\. Step 1", "", " ".join(exp.split()))).ratio()
        else:
            same = sent == " ".join(exp.split())
            ratio = difflib.SequenceMatcher(None, sent, " ".join(exp.split())).ratio()
        diff = ""
        if not same:
            sm = difflib.SequenceMatcher(None, sent, " ".join(exp.split()))
            ops = [(t, sent[a:b], " ".join(exp.split())[c2:d2]) for t, a, b, c2, d2 in sm.get_opcodes() if t != "equal"][:3]
            diff = f"similarity {ratio:.3f}; " + " / ".join(f"{t}: sent [{x[:60]}] expected [{y[:60]}]" for t, x, y in ops)
        c.append(chk(f"{mode} prompt copied exactly (call {k + 1})", same, diff))
        c.append(chk(f"{mode} max tokens {tokens} (call {k + 1})", f"max tokens {tokens}" in ins))
    return c


# --------------------------------------------------------------------------- main


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--runs", type=int, default=2)
    ap.add_argument("--start-run", type=int, default=1)
    ap.add_argument("--only", default="")
    ap.add_argument("--budget", type=int, default=120)
    ap.add_argument("--workers", type=int, default=4)
    ap.add_argument("--replay", default="", help="recorded run json; replay its first --upto assistant turns")
    ap.add_argument("--upto", type=int, default=0)
    ap.add_argument("--as-run", type=int, default=0)
    ap.add_argument("--reeval", action="store_true", help="recompute checks of saved runs, no API calls")
    args = ap.parse_args()
    if args.reeval:
        rows = []
        for f in sorted(OUT.glob("S*_r*.json")):
            lg = json.loads(f.read_text(encoding="utf-8"))
            lg["checks"] = evaluate(lg["scenario"], lg)
            lg["pass"] = all(x["ok"] for x in lg["checks"] if x["critical"])
            f.write_text(json.dumps(lg, ensure_ascii=False, indent=1), encoding="utf-8")
            print(f.name, "PASS" if lg["pass"] else "FAIL", [x["name"] for x in lg["checks"] if not x["ok"]])
            rows.append({"scenario": lg["scenario"], "run": lg["run"], "pass": lg["pass"], "calls": lg["calls"],
                         "replayed_turns": lg.get("replayed_turns", 0), "steps": [d["step"] for d in lg["delegates"]],
                         "failed_critical": [x["name"] for x in lg["checks"] if not x["ok"] and x["critical"]],
                         "failed_minor": [x["name"] for x in lg["checks"] if not x["ok"] and not x["critical"]]})
        (OUT / "summary.json").write_text(json.dumps({"runs": rows}, ensure_ascii=False, indent=1), encoding="utf-8")
        return
    if args.replay:
        BUDGET["left"] = args.budget
        system_prompt, _ = build_system_prompt()
        rec = json.loads(Path(args.replay).read_text(encoding="utf-8"))
        texts = [t["content"] for t in rec["turns"] if t["role"] == "assistant"][: args.upto]
        lg = run_scenario(rec["scenario"], args.as_run or rec["run"], system_prompt, replay=texts)
        print(lg["scenario"], lg["run"], "PASS" if lg["pass"] else "FAIL", "calls", lg["calls"],
              [d["step"] for d in lg["delegates"]], [x["name"] for x in lg["checks"] if not x["ok"]])
        return
    BUDGET["left"] = args.budget
    system_prompt, linked = build_system_prompt()
    print(f"system prompt: {len(system_prompt)} chars, linked files: {', '.join(linked)}")
    ids = [s for s in SCENARIOS if not args.only or s in args.only.split(",")]
    jobs = [(s, r) for r in range(args.start_run, args.runs + 1) for s in ids]
    with ThreadPoolExecutor(max_workers=args.workers) as ex:
        futs = [ex.submit(run_scenario, s, r, system_prompt) for s, r in jobs]
        logs = []
        for fu in futs:
            try:
                logs.append(fu.result())
            except Exception as exn:  # keep the other runs
                print("run crashed:", type(exn).__name__, exn)
    summary = []
    for lg in logs:
        failed = [x["name"] for x in lg["checks"] if not x["ok"]]
        summary.append({"scenario": lg["scenario"], "run": lg["run"], "pass": lg["pass"], "calls": lg["calls"],
                        "steps": [d["step"] for d in lg["delegates"]], "failed_checks": failed, "error": lg["error"]})
        print(f"{lg['scenario']} r{lg['run']}: {'PASS' if lg['pass'] else 'FAIL'} calls={lg['calls']} steps={summary[-1]['steps']}")
        for x in lg["checks"]:
            if not x["ok"]:
                print(f"   - {'!' if x['critical'] else '~'} {x['name']}: {x['detail'][:300]}")
    (OUT / "summary.json").write_text(json.dumps({"api_calls": BUDGET["used"], "runs": summary}, ensure_ascii=False, indent=1), encoding="utf-8")
    print(f"API calls used: {BUDGET['used']}")


if __name__ == "__main__":
    main()
